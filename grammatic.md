<statement> ::= <объявлениепеременной> | <присваивание> | <ввод> | <вывод> | <if> | <while> | <for> | <вызовинструкция> | <return>

<тип> ::= <типэлемента> | "цепь" "[" <int> "]" "из" <типэлемента>

<if> ::= "Если истинно предзнаменование" "(" <expression> ")" "," "да свершится следующее" <конецстроки>
         <блок>
         [ "Если же нет" "," "да свершится иное" <конецстроки> <блок> ]
         "Да свершится предначертанное" <конецстроки>

<while> ::= "О великие силы повторения" "," "взываю к вам" <конецстроки>
            "Покуда истинно предзнаменование" "(" <expression> ")" "," "вершите" <конецстроки>
            <блок>
            "Да замкнётся круг" <конецстроки>

<for> ::= "О хранители пути" "," "проведите" <name>
          "от" "(" <expression> ")" "до" "(" <expression> ")"
          "с шагом" "(" <expression> ")" <конецстроки>
          <блок>
          "Да завершится шествие" <конецстроки>

<структура> ::= "Наречём новый облик" <name> <конецстроки> <поле> { <поле> } "Да завершится облик" <конецстроки>
<поле> ::= <name> <тип> <конецстроки>
<перечисление> ::= "Наречём перечень" <name> <конецстроки> <name> { "," <name> } <конецстроки> "Да завершится перечень" <конецстроки>
<созданиеструктуры> ::= "воплощение" <name> "(" <аргументы> ")"

<функция> ::= "Запишем заклинание" <name> "с дарами" "(" <параметры> ")" ","
              "возвращающее" <типвозврата> <конецстроки> <блок> "Да запечатается заклинание" <конецстроки>
<параметры> ::= [ <параметр> { ";" <параметр> } ]
<параметр> ::= <name> <тип>
<аргументы> ::= [ <expression> { ";" <expression> } ]
<вызоввыражение> ::= "Исполнение заклинания " <name> "с дарами" "(" <аргументы> ")"
<вызовинструкция> ::= "Да исполнится" <name> "с дарами" "(" <аргументы> ")" <конецстроки>
<return> ::= "Да станет ответом" <expression> <конецстроки> | "Да завершится призыв" <конецстроки>

<expression> ::= <expression13>

<expression11> ::= <expression10> { <operation11> <expression10> }
<expression10> ::= <expression9> { <operation10> <expression9> }
<expression9> ::= <expression8> { <operation9> <expression8> }
<expression8> ::= <expression7> { <operation8> <expression7> }
<expression7> ::= <expression6> [ <operation7> <expression6> ]
<expression6> ::= <expression5> [ <operation6> <expression5> ]
<expression5> ::= <expression4> { <operation5> <expression4> }
<expression4> ::= <expression3> { <operation4> <expression3> }
<expression3> ::= <expression2> { <operation3> <expression2> }
<expression2> ::= <operation2> <expression2> | <expression1>
<expression1> ::= <основавыражения> { <индексилиполе> }

<operation13> ::= ","

<operation11> ::= "&&"

          | "а" | "б" | "в" | "г" | "д" | "е" | "ё" | "ж" | "з" | "и" | "й" | "к" | "л" | "м" | "н" | "о" | "п" | "р" | "с" | "т" | "у" | "ф" | "х" | "ц" | "ч" | "ш" | "щ" | "ъ" | "ы" | "ь" | "э" | "ю" | "я"

<дробное> ::= <int> "." <int>

<спецсимвол> ::= "\" ( "\n" | "\t" | "\" | "'" | '"' )
